cmake_minimum_required(VERSION 3.16)
project(chronobent_installed_consumers LANGUAGES C CXX)
find_package(chronobent 0.2 CONFIG REQUIRED)
enable_testing()
add_executable(consumer-c ../../examples/processor_c.c)
set_target_properties(consumer-c PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(consumer-c PRIVATE chronobent::chronobent)
add_test(NAME installed-c COMMAND consumer-c)
add_executable(consumer-cpp ../../examples/processor_cpp.cpp)
target_compile_features(consumer-cpp PRIVATE cxx_std_17)
target_link_libraries(consumer-cpp PRIVATE chronobent::chronobent)
add_test(NAME installed-cpp COMMAND consumer-cpp)
add_executable(consumer-legacy ../test_c.c)
set_target_properties(consumer-legacy PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(consumer-legacy PRIVATE chronobent::chronobent)
add_test(NAME installed-legacy-c COMMAND consumer-legacy)
set(consumer_tests installed-c installed-cpp installed-legacy-c)
get_target_property(consumer_definitions chronobent::chronobent INTERFACE_COMPILE_DEFINITIONS)
if("CHRONOBENT_HAS_TUNING=1" IN_LIST consumer_definitions)
    add_executable(consumer-tuning-c ../test_tuning_c.c)
    set_target_properties(consumer-tuning-c PROPERTIES LINKER_LANGUAGE CXX)
    target_link_libraries(consumer-tuning-c PRIVATE chronobent::chronobent)
    add_test(NAME installed-tuning-c COMMAND consumer-tuning-c)
    add_executable(consumer-tuning-cpp ../../examples/tuning_cpp.cpp)
    target_compile_features(consumer-tuning-cpp PRIVATE cxx_std_17)
    target_link_libraries(consumer-tuning-cpp PRIVATE chronobent::chronobent)
    add_test(NAME installed-tuning-cpp COMMAND consumer-tuning-cpp)
    list(APPEND consumer_tests installed-tuning-c installed-tuning-cpp)
endif()
# Windows does not embed an rpath. The test environment supplies the installed
# DLL directory without copying the build-tree DLL or adding source includes.
if(WIN32)
    get_target_property(consumer_dll chronobent::chronobent IMPORTED_LOCATION_RELEASE)
    if(consumer_dll)
        get_filename_component(consumer_dll_dir "${consumer_dll}" DIRECTORY)
        string(REPLACE ";" "\\;" consumer_path "${consumer_dll_dir};$ENV{PATH}")
        set_tests_properties(${consumer_tests} PROPERTIES
            ENVIRONMENT "PATH=${consumer_path}")
    endif()
endif()
